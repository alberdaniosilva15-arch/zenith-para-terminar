Este documento prova que TODAS as RPCs críticas agora usam auth.uid()
em vez de parâmetros client-controlled (p_driver_id / p_user_id).

Cada secção mostra: (1) o SQL real na BD, (2) o frontend que chama a RPC,
(3) a verificação da BD de produção.

1. accept_ride_atomic — ANTES vs DEPOIS
================================================================================

--- ANTES (VULNERÁVEL) ---
Ficheiro: migrations/20260407_accept_ride_atomic.sql

  CREATE OR REPLACE FUNCTION public.accept_ride_atomic(
    p_ride_id   UUID,
    p_driver_id UUID          <-- CLIENT CONTROLA ISTO!
  )
  ...
  UPDATE public.rides
  SET driver_id = p_driver_id,    <-- injeta qualquer ID
      status = 'accepted',
      accepted_at = NOW()
  WHERE id = p_ride_id;

--- DEPOIS (SEGURO) ---
Ficheiro: supabase/migrations/20260719_rpc_security_fix_auth_uid.sql

  DROP FUNCTION IF EXISTS public.accept_ride_atomic(UUID, UUID);
  CREATE OR REPLACE FUNCTION public.accept_ride_atomic(
    p_ride_id UUID              <-- SEM p_driver_id
  )
  RETURNS JSONB
  LANGUAGE plpgsql
  SECURITY DEFINER
  SET search_path = pg_catalog, public
  AS $$
  DECLARE
    v_driver_id UUID := auth.uid();   <-- USA auth.uid() DIRECTAMENTE
    v_ride      RECORD;
  BEGIN
    IF v_driver_id IS NULL THEN
      RETURN jsonb_build_object('success', false, 'reason', 'not_authenticated');
    END IF;

    IF NOT public.is_driver() THEN
      RETURN jsonb_build_object('success', false, 'reason', 'not_a_driver');
    END IF;

    SELECT * INTO v_ride
    FROM public.rides
    WHERE id = p_ride_id
    FOR UPDATE NOWAIT;
    ...

    UPDATE public.rides
    SET driver_id = v_driver_id,   <-- auth.uid(), NÃO parâmetro do cliente
        status = 'accepted',
        accepted_at = NOW(),
        driver_confirmed = TRUE,
        updated_at = NOW()
    WHERE id = p_ride_id;
  ...

--- Frontend que chama (rideService.ts) ---
  // ANTES:
  const { data, error } = await supabase.rpc('accept_ride_atomic', {
    p_ride_id:   rideId,
    p_driver_id: driverId,    <-- passava o ID manualmente
  });

  // DEPOIS:
  const { data, error } = await supabase.rpc('accept_ride_atomic', {
    p_ride_id: rideId,        <-- SÓ o ride_id, sem driver_id
  });

--- Verificação BD produção ---

    FROM pg_proc WHERE proname = 'accept_ride_atomic';"

3. cancel_ride_safe — ANTES vs DEPOIS
================================================================================

--- ANTES (VULNERÁVEL) ---
Ficheiro: zenith_schema_final.sql

  CREATE OR REPLACE FUNCTION public.cancel_ride_safe(
    p_ride_id UUID,
    p_user_id UUID,            <-- CLIENT CONTROLA ISTO!
    p_reason  TEXT DEFAULT 'Cancelado'
  )
  ...
  IF v_ride.passenger_id <> p_user_id
    AND (v_ride.driver_id IS NULL OR v_ride.driver_id <> p_user_id)
  THEN
    RETURN QUERY SELECT false, v_prev_status, 'Sem permissão.'::TEXT;
  END IF;
  ...
  -- Qualquer um podia cancelar corridas de outro!

--- DEPOIS (SEGURO) ---
  DROP FUNCTION IF EXISTS public.cancel_ride_safe(UUID, UUID, TEXT);
  CREATE OR REPLACE FUNCTION public.cancel_ride_safe(
    p_ride_id UUID,
    p_reason  TEXT DEFAULT 'Cancelado pelo utilizador'  <-- SEM p_user_id
  )
  ...
  DECLARE
    v_user_id UUID := auth.uid();   <-- USA auth.uid()
  ...
  IF v_user_id IS NULL THEN
    RETURN QUERY SELECT false, 'unknown'::TEXT, 'Autenticação necessária.'::TEXT;
  END IF;
  ...
  IF v_ride.passenger_id <> v_user_id
    AND (v_ride.driver_id IS NULL OR v_ride.driver_id <> v_user_id)
  THEN
    RETURN QUERY SELECT false, v_prev_status, 'Sem permissão para cancelar.'::TEXT;
  END IF;

--- Frontend (rideService.ts) ---
  // ANTES:

    p_ride_id: rideId,
    p_user_id: userId,        <-- passava o ID do user
    p_reason: reason ?? 'Cancelado',
  });

  // DEPOIS:
  const { data, error } = await supabase.rpc('cancel_ride_safe', {
    p_ride_id: rideId,        <-- SÓ o ride_id
    p_reason: reason ?? 'Cancelado pelo utilizador',
  });

      case RideStatus.PICKING_UP:

  -- driver_locations: QUALQUER UM pode ver TODOS os GPS!
  CREATE POLICY "driver_locations_public_read"
  ON public.driver_locations FOR SELECT
  USING (true);                  <-- EXPÕE GPS DE TODOS OS MOTORISTAS

  -- rides: QUALQUER UM pode ver TODAS as corridas!
  CREATE POLICY "rides: motorista vê TODAS no mapa"
  ON public.rides FOR SELECT
  USING (true);                  <-- EXPÕE DADOS DE TODAS AS CORRIDAS

--- DEPOIS (migration 20260719) ---
  -- rides: passageiro vê só as suas
  CREATE POLICY "rides_select_own_passenger"
    ON public.rides FOR SELECT TO authenticated
    USING (passenger_id = auth.uid() AND public.is_passenger());

  -- rides: motorista vê só as suas
  CREATE POLICY "rides_select_own_driver"
    ON public.rides FOR SELECT TO authenticated
    USING (driver_id = auth.uid() AND public.is_driver());

  -- driver_locations: motorista vê só a sua
  CREATE POLICY "dl_select_own_driver"
    ON public.driver_locations FOR SELECT TO authenticated
    USING (driver_id = auth.uid() AND public.is_driver());

8. ANON BLOQUEADO
================================================================================

--- Verificação: permissões do role 'anon' nas tabelas sensíveis ---
  supabase db query --linked "SELECT grantee, table_name, privilege_type
    FROM information_schema.role_table_grants
    WHERE grantee = 'anon'
    AND table_name IN ('rides', 'driver_locations');"

  (nenhum resultado)  <-- ZERO permissões para anon

--- SQL que garante isso (migration 20260719) ---
  ALTER TABLE public.rides FORCE ROW LEVEL SECURITY;
  ALTER TABLE public.driver_locations FORCE ROW LEVEL SECURITY;

  REVOKE ALL ON public.rides FROM anon;
  REVOKE ALL ON public.driver_locations FROM anon;
